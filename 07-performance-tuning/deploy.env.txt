# Replication slot name -- tied to the standby identity so it is consistent
# on both primary and standby. Do NOT use hostname of the local machine.
REP_SLOT_NAME="standby_$(echo "$STANDBY_HOST" | tr '.' '_')"

# Boolean: must be compared as string ( = "true" ). Anything else disables.
USE_REPLICATION_SLOT="false"

# TDE Configuration -- passphrase is injected at runtime via environment.
# Export TDE_PASSPHRASE before sourcing this file. Never commit a real key.
TDE_PASSPHRASE="${TDE_PASSPHRASE:-}"
TDE_WRAP_CMD="openssl enc -e -aes-128-cbc -pbkdf2 -pass pass:${TDE_PASSPHRASE} -out %p"
TDE_UNWRAP_CMD="openssl enc -d -aes-128-cbc -pbkdf2 -pass pass:${TDE_PASSPHRASE} -in %p"
